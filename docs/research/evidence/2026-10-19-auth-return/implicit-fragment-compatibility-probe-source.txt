import React from 'react'
import { readFileSync } from 'node:fs'
import path from 'node:path'
import ts from 'typescript'
import { render, waitFor, screen } from '@testing-library/react'
import CurrentSignIn from '@/app/[locale]/auth/signin/page'

const mockReplace = jest.fn()
const mockPush = jest.fn()
const mockSession = jest.fn()
const mockSearch = new URLSearchParams('error=auth_failed&next=%2Fen%2Fdashboard')
jest.mock('next/navigation', () => ({ useRouter: () => ({ push: mockPush, replace: mockReplace }), useSearchParams: () => mockSearch }))
jest.mock('next-intl', () => ({ useLocale: () => 'en', useTranslations: () => (key: string) => key }))
jest.mock('@/lib/supabase', () => ({ createClientSupabase: () => ({ auth: { setSession: mockSession } }) }))

// Recovery-only probe. The old implementation is compiled from immutable before bytes;
// no current application file is swapped, and no provider/browser network is used.
test('compatibility change: fixed callback error plus inherited implicit fragment fails closed now', async () => {
  const before = readFileSync(path.resolve(process.cwd(), '../../.superpowers/sdd/2026-10-19-upgrade-auth-return/before/packages/web/src/app/[locale]/auth/signin/page.tsx'), 'utf8')
  const compiled = ts.transpileModule(before, { compilerOptions: { module: ts.ModuleKind.CommonJS, jsx: ts.JsxEmit.ReactJSX, esModuleInterop: true } }).outputText
  const recovered: { exports: { default?: React.ComponentType } } = { exports: {} }
  new Function('require', 'module', 'exports', compiled)(require, recovered, recovered.exports)
  const OldSignIn = recovered.exports.default!
  const user = { id: '11111111-1111-4111-8111-111111111111' }
  const session = { access_token: 'synthetic-access', refresh_token: 'synthetic-refresh', token_type: 'bearer', expires_in: 3600, expires_at: 2_000_000_000, user }
  mockSession.mockResolvedValue({ data: { session, user }, error: null })
  const navigate = () => window.history.replaceState(null, '', '/en/auth/signin?error=auth_failed&next=%2Fen%2Fdashboard#access_token=synthetic-access&refresh_token=synthetic-refresh&token_type=bearer')
  const log = jest.spyOn(console, 'log').mockImplementation(() => {})
  const error = jest.spyOn(console, 'error').mockImplementation(() => {})
  try {
    navigate()
    const beforeView = render(<OldSignIn />)
    await waitFor(() => expect(mockPush).toHaveBeenCalledWith('/en/dashboard'))
    const beforeResult = { setSessionCalls: mockSession.mock.calls.length, navigations: mockPush.mock.calls.length }
    expect(beforeResult).toEqual({ setSessionCalls: 1, navigations: 1 })
    beforeView.unmount()
    mockSession.mockClear(); mockPush.mockClear(); mockReplace.mockClear()
    navigate()
    const afterView = render(<CurrentSignIn />)
    expect(await screen.findByText('Error: Authentication could not be confirmed. Please sign in again.')).toBeInTheDocument()
    const afterResult = { setSessionCalls: mockSession.mock.calls.length, navigations: mockReplace.mock.calls.length + mockPush.mock.calls.length, cleanedHash: window.location.hash }
    expect(afterResult).toEqual({ setSessionCalls: 0, navigations: 0, cleanedHash: '' })
    process.stdout.write(JSON.stringify({ scenario: 'fixed callback error + implicit fragment', before: beforeResult, current: afterResult, classification: 'scoped compatibility change / release blocker; no real provider or browser redirect integration claimed' }, null, 2) + '\n')
    afterView.unmount()
  } finally { log.mockRestore(); error.mockRestore() }
})
